# Gazebo 월드 & WorldPlugin Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Gazebo Classic 11 원룸 월드(벽·가구·충전스테이션·로봇·사람 placeholder)와, 로봇/사람의 ground-truth 물리 데이터를 ROS2 토픽으로 발행하고 사람의 활동(걷기/낙상 등)을 커맨드로 조종할 수 있는 WorldPlugin을 구현한다. `설계안_가제보시뮬레이션_관리자콘솔.md` §3, §5, §6의 시뮬레이션 레이어에 해당한다.

**Architecture:** 로봇 이동은 표준 `gazebo_ros diff_drive`/`camera` 플러그인을 그대로 재사용(직접 구현하지 않음). 커스텀 `happy_care_world_plugin`은 딱 두 책임만 가진다 — ① 매 스텝 로봇 pose·도킹거리·사람 pose/속도를 ROS2 토픽으로 발행, ② `/happy_care/actor_cmd` 커맨드를 받아 사람 모델의 포즈를 보간 전환. 활동 분류·모드 전환 판단 로직은 이 플랜에 넣지 않는다 — 다음 플랜(백엔드 연동)의 책임이다.

**Tech Stack:** Gazebo Classic 11, ROS 2 Humble, gazebo_ros, C++17, ament_cmake, Python(launch)

## Global Constraints

- Gazebo **Classic 11**만 사용 (신세대 Gazebo/ros_gz_bridge 아님 — `/usr/bin/gz`는 이미 classic 소유, 충돌 방지)
- 폴더는 `sim/`을 레포 최상위에 평탄하게 (`backend/`, `web/`, `docs/`와 동일 레벨)
- 로봇 `/cmd_vel`·카메라는 표준 `libgazebo_ros_diff_drive.so`/`libgazebo_ros_camera.so` 재사용 — 커스텀 코드로 재구현하지 않는다
- WorldPlugin은 판단 로직(활동 분류·모드 디바운스)을 갖지 않는다 — ground-truth 발행 + 커맨드 반영만
- 이 도메인은 pytest 스타일 유닛테스트가 성립하지 않는다(Gazebo 프로세스 필요) — 각 스텝의 "테스트"는 `colcon build` 성공 + headless 기동 후 `ros2 topic echo`/`ros2 topic pub` CLI로 실제 토픽 값을 확인하는 것으로 대체한다
- 빌드는 콜콘 워크스페이스가 필요 — 최초 1회만: `mkdir -p ~/happy_care_ws/src && ln -s ~/happy-care/sim ~/happy_care_ws/src/happy_care_sim`. 이후 모든 빌드는 `cd ~/happy_care_ws && colcon build --packages-select happy_care_sim && source install/setup.bash`

**최종 ROS2 토픽 계약 (다음 플랜이 이 값을 그대로 구독한다):**

| 토픽 | 타입 | 발행/구독 | 의미 |
|---|---|---|---|
| `/happy_care/robot/pose` | `geometry_msgs/Pose` | 발행 | 로봇 실측 pose |
| `/happy_care/robot/dock_distance` | `std_msgs/Float64` | 발행 | 로봇↔충전스테이션 거리(m) |
| `/happy_care/robot/cmd_vel` | `geometry_msgs/Twist` | 구독(표준 plugin) | 텔레옵 속도 명령 |
| `/happy_care/robot/camera/image_raw` | `sensor_msgs/Image` | 발행(표준 plugin) | 로봇 1인칭 카메라 |
| `/happy_care/person/pose` | `geometry_msgs/Pose` | 발행 | 사람(액터) 실측 pose |
| `/happy_care/person/torso_speed` | `std_msgs/Float64` | 발행 | 사람 이동 속도(m/s) |
| `/happy_care/actor_cmd` | `std_msgs/String` | 구독 | `idle`/`sit`/`lie`/`fall`/`recover` |

                '-topic', '/happy_care/robot/robot_description',

            ],
            output='screen',
        ),
    ])

- [ ] **Step 1: WorldPlugin 소스 작성 (ground-truth 발행만, 사람은 아직 고정)**

`sim/src/world_plugin.cpp`:
```cpp
#include <gazebo/common/Plugin.hh>
#include <gazebo/physics/physics.hh>
#include <gazebo_ros/node.hpp>
#include <rclcpp/rclcpp.hpp>
#include <geometry_msgs/msg/pose.hpp>
#include <std_msgs/msg/float64.hpp>

namespace happy_care_sim
{

class HappyCareWorldPlugin : public gazebo::WorldPlugin
{
public:
  void Load(gazebo::physics::WorldPtr world, sdf::ElementPtr sdf) override
  {
    world_ = world;
    ros_node_ = gazebo_ros::Node::Get(sdf);

    robot_model_name_ = sdf->Get<std::string>("robot_model", "haengbogi").first;
    station_model_name_ = sdf->Get<std::string>("station_model", "charging_station").first;
    person_model_name_ = sdf->Get<std::string>("person_model", "person").first;

    robot_pose_pub_ = ros_node_->create_publisher<geometry_msgs::msg::Pose>(
      "/happy_care/robot/pose", 10);
    dock_distance_pub_ = ros_node_->create_publisher<std_msgs::msg::Float64>(
      "/happy_care/robot/dock_distance", 10);
    person_pose_pub_ = ros_node_->create_publisher<geometry_msgs::msg::Pose>(
      "/happy_care/person/pose", 10);
    person_speed_pub_ = ros_node_->create_publisher<std_msgs::msg::Float64>(
      "/happy_care/person/torso_speed", 10);

    last_update_time_ = world_->SimTime();
    auto person = world_->ModelByName(person_model_name_);
    if (person) {
      last_person_pos_ = person->WorldPose().Pos();
    }

    update_connection_ = gazebo::event::Events::ConnectWorldUpdateBegin(
      std::bind(&HappyCareWorldPlugin::OnUpdate, this));

    RCLCPP_INFO(ros_node_->get_logger(), "happy_care_world_plugin loaded");
  }

private:
  static geometry_msgs::msg::Pose ToMsg(const ignition::math::Pose3d & p)
  {
    geometry_msgs::msg::Pose msg;
    msg.position.x = p.Pos().X();
    msg.position.y = p.Pos().Y();
    msg.position.z = p.Pos().Z();
    msg.orientation.x = p.Rot().X();
    msg.orientation.y = p.Rot().Y();
    msg.orientation.z = p.Rot().Z();
    msg.orientation.w = p.Rot().W();
    return msg;
  }

  void OnUpdate()
  {
    auto now = world_->SimTime();
    double dt = (now - last_update_time_).Double();
    if (dt <= 0) {
      return;
    }

    auto robot = world_->ModelByName(robot_model_name_);
    auto station = world_->ModelByName(station_model_name_);
    if (robot && station) {
      auto robot_pose = robot->WorldPose();
      robot_pose_pub_->publish(ToMsg(robot_pose));

      std_msgs::msg::Float64 dist_msg;
      dist_msg.data = robot_pose.Pos().Distance(station->WorldPose().Pos());
      dock_distance_pub_->publish(dist_msg);
    }

    auto person = world_->ModelByName(person_model_name_);
    if (person) {
      auto person_pose = person->WorldPose();
      person_pose_pub_->publish(ToMsg(person_pose));

      std_msgs::msg::Float64 speed_msg;
      speed_msg.data = (person_pose.Pos() - last_person_pos_).Length() / dt;
      person_speed_pub_->publish(speed_msg);
      last_person_pos_ = person_pose.Pos();
    }

    last_update_time_ = now;
  }

  gazebo::physics::WorldPtr world_;
  gazebo_ros::Node::SharedPtr ros_node_;
  gazebo::event::ConnectionPtr update_connection_;

  std::string robot_model_name_, station_model_name_, person_model_name_;

  rclcpp::Publisher<geometry_msgs::msg::Pose>::SharedPtr robot_pose_pub_;
  rclcpp::Publisher<std_msgs::msg::Float64>::SharedPtr dock_distance_pub_;
  rclcpp::Publisher<geometry_msgs::msg::Pose>::SharedPtr person_pose_pub_;
  rclcpp::Publisher<std_msgs::msg::Float64>::SharedPtr person_speed_pub_;

  gazebo::common::Time last_update_time_;
  ignition::math::Vector3d last_person_pos_{0, 0, 0};
};

GZ_REGISTER_WORLD_PLUGIN(HappyCareWorldPlugin)

}  // namespace happy_care_sim
```

import os
from launch import LaunchDescription
from launch.actions import IncludeLaunchDescription
from launch.launch_description_sources import PythonLaunchDescriptionSource
from launch_ros.actions import Node
from ament_index_python.packages import get_package_share_directory


def generate_launch_description():
    pkg_share = get_package_share_directory('happy_care_sim')
    world_path = os.path.join(pkg_share, 'worlds', 'happy_care_room.world')
    urdf_path = os.path.join(pkg_share, 'urdf', 'haengbogi.urdf')

    with open(urdf_path, 'r') as f:
        robot_description = f.read()

    gazebo = IncludeLaunchDescription(
        PythonLaunchDescriptionSource(
            os.path.join(
                get_package_share_directory('gazebo_ros'),
                'launch', 'gazebo.launch.py')
        ),
        launch_arguments={'world': world_path}.items()
    )

    robot_state_publisher = Node(
        package='robot_state_publisher',
        executable='robot_state_publisher',
        namespace='happy_care/robot',
        parameters=[{'robot_description': robot_description}],
    )

    spawn_robot = Node(
        package='gazebo_ros',
        executable='spawn_entity.py',
        arguments=[
            '-entity', 'haengbogi',
            '-topic', '/happy_care/robot/robot_description',
            '-x', '0', '-y', '0', '-z', '0.2',
        ],
        output='screen',
    )

    web_video_server = Node(
        package='web_video_server',
        executable='web_video_server',
        parameters=[{'port': 8080}],
    )

    return LaunchDescription([
        gazebo,
        robot_state_publisher,
        spawn_robot,
        web_video_server,
    ])
